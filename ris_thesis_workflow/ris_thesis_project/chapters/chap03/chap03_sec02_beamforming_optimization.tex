\section{基于交替优化与分式规划的主被动波束协同设计}
\label{sec:beamforming_optimization}

针对多用户 MISO 和速率优化问题~\eqref{prob:joint_beamforming_p1}中发射预编码矩阵 $\bm{W}$ 与无源相移向量 $\bm{v}$ 的高度非凸非线性耦合，本节提出一套基于交替优化（Alternating Optimization, AO）与多比率分式规划（Fractional Programming, FP）的高效解耦协同优化算法。

\subsection{基于二次变换的多比率分式规划解耦}
由于目标函数中 SINR 处于对数函数内部且各项互为干扰，传统 Dinkelbach 标量分式规划无法直接处理多比率求和问题。在此，引入基于拉格朗日对偶的二次变换技术（Quadratic Transform）。
定义辅助变量 $\bm{\rho} = [\rho_1, \rho_2, \dots, \rho_K]^T \in \mathbb{R}_+^K$ 与 $\bm{\mu} = [\mu_1, \mu_2, \dots, \mu_K]^T \in \mathbb{C}^{K \times 1}$，根据分式规划等价转化定理，问题~\eqref{prob:joint_beamforming_p1}等价重构为如下增广目标函数的极大化问题：
\begin{equation}
  f_q(\bm{W}, \bm{v}, \bm{\rho}, \bm{\mu}) \triangleq \sum_{k=1}^K \left[ \log_2(1 + \rho_k) - \rho_k + \frac{(1+\rho_k) \cdot 2\mathrm{Re}\{\mu_k^* \bm{h}_k^H(\bm{v})\bm{w}_k\}}{1 + \gamma_k(\bm{W}, \bm{v})} - |\mu_k|^2 \left( \sum_{j=1}^K |\bm{h}_k^H(\bm{v})\bm{w}_j|^2 + \sigma_k^2 \right) \right]
  \label{eq:fp_augmented_objective}
\end{equation}
通过对辅助变量求偏导并令其为零，可得辅助变量 $\bm{\rho}$ 与 $\bm{\mu}$ 在固定 $\bm{W}$ 与 $\bm{v}$ 时的解析闭式最优更新：
\begin{align}
  \rho_k^* &= \gamma_k(\bm{W}, \bm{v}) = \frac{|\bm{h}_k^H(\bm{v})\bm{w}_k|^2}{\sum_{j \neq k}^K |\bm{h}_k^H(\bm{v})\bm{w}_j|^2 + \sigma_k^2} \label{eq:fp_update_rho} \\
  \mu_k^* &= \frac{\sqrt{1 + \rho_k^*} \bm{h}_k^H(\bm{v})\bm{w}_k}{\sum_{j=1}^K |\bm{h}_k^H(\bm{v})\bm{w}_j|^2 + \sigma_k^2} \label{eq:fp_update_mu}
\end{align}

\subsection{子问题一：固定相移求解基站有源预编码 \texorpdfstring{$\bm{W}$}{W}}
在固定 $\bm{v}$、$\bm{\rho}$ 与 $\bm{\mu}$ 的条件下，目标函数式~\eqref{eq:fp_augmented_objective}关于基站有源预编码矩阵 $\bm{W}$ 退化为一个标准的凸二次规划（QP）问题：
\begin{subequations}
  \label{prob:sub1_w_opt}
  \begin{align}
    \min_{\bm{W}} \quad & \sum_{j=1}^K \bm{w}_j^H \left( \sum_{k=1}^K |\mu_k|^2 \bm{h}_k(\bm{v})\bm{h}_k^H(\bm{v}) \right) \bm{w}_j - 2\sum_{k=1}^K \sqrt{1+\rho_k} \mathrm{Re}\left\{ \mu_k^* \bm{h}_k^H(\bm{v})\bm{w}_k \right\} \label{prob:sub1_obj} \\
    \text{s.t.} \quad & \sum_{k=1}^K \|\bm{w}_k\|^2 \le P_{\max} \label{prob:sub1_power}
  \end{align}
\end{subequations}

利用拉格朗日乘子法，引入对偶乘子 $\lambda \ge 0$ 对应发射功率约束~\eqref{prob:sub1_power}，构造拉格朗日函数 $\mathcal{L}(\bm{W}, \lambda)$。对任意用户 $k$ 的波束赋形向量 $\bm{w}_k$ 求复梯度并令其为零：
\begin{equation}
  \nabla_{\bm{w}_k^*} \mathcal{L} = \left( \sum_{j=1}^K |\mu_j|^2 \bm{h}_j(\bm{v})\bm{h}_j^H(\bm{v}) + \lambda \bm{I}_M \right) \bm{w}_k - \sqrt{1+\rho_k}\mu_k \bm{h}_k(\bm{v}) = \bm{0}
  \label{eq:w_k_derivative}
\end{equation}
由此可得最优发射预编码向量 $\bm{w}_k^*$ 的显式解析闭式解：
\begin{equation}
  \bm{w}_k^*(\lambda) = \left( \bm{A}(\bm{v}, \bm{\mu}) + \lambda \bm{I}_M \right)^{-1} \sqrt{1+\rho_k}\mu_k \bm{h}_k(\bm{v})
  \label{eq:optimal_wk_closed_form}
\end{equation}
其中 $\bm{A}(\bm{v}, \bm{\mu}) \triangleq \sum_{j=1}^K |\mu_j|^2 \bm{h}_j(\bm{v})\bm{h}_j^H(\bm{v}) \in \mathbb{C}^{M \times M}$ 为半正定等效协方差矩阵。对偶乘子 $\lambda \ge 0$ 可通过一维二分搜索法（Bisection Method）在区间 $[0, \lambda_{\max}]$ 内精确确定，满足互补松弛条件：
\begin{equation}
  \lambda \left( \sum_{k=1}^K \|\bm{w}_k^*(\lambda)\|^2 - P_{\max} \right) = 0
  \label{eq:slackness_condition}
\end{equation}

\subsection{子问题二：固定预编码求解 RIS 无源相移向量 \texorpdfstring{$\bm{v}$}{v}}
在给定基站预编码矩阵 $\bm{W}$ 及辅助变量 $\bm{\rho}, \bm{\mu}$ 的条件下，将复合信道 $\bm{h}_k^H(\bm{v}) = \bm{h}_{d,k}^H + \bm{v}^H \bm{H}_{\mathrm{casc},k}$ 代入目标函数式~\eqref{eq:fp_augmented_objective}中展开。
经代数重组，关于相移向量 $\bm{v} \in \mathbb{C}^{N \times 1}$ 的优化问题可等价整理为标准的复二次型极大化问题：
\begin{subequations}
  \label{prob:sub2_v_opt}
  \begin{align}
    \min_{\bm{v}} \quad & \bm{v}^H \bm{\Psi} \bm{v} - 2\mathrm{Re}\{\bm{v}^H \bm{q}\} \label{prob:sub2_obj} \\
    \text{s.t.} \quad & |v_n| = 1, \quad \forall n = 1, 2, \dots, N \label{prob:sub2_unimodular}
  \end{align}
\end{subequations}
其中矩阵 $\bm{\Psi} \in \mathbb{C}^{N \times N}$ 与向量 $\bm{q} \in \mathbb{C}^{N \times 1}$ 分别定义为：
\begin{align}
  \bm{\Psi} &\triangleq \sum_{k=1}^K |\mu_k|^2 \sum_{j=1}^K \bm{H}_{\mathrm{casc},k} \bm{w}_j \bm{w}_j^H \bm{H}_{\mathrm{casc},k}^H \label{eq:psi_matrix_def} \\
  \bm{q} &\triangleq \sum_{k=1}^K \left[ \sqrt{1+\rho_k}\mu_k \bm{H}_{\mathrm{casc},k}\bm{w}_k - |\mu_k|^2 \sum_{j=1}^K \bm{H}_{\mathrm{casc},k} \bm{w}_j \bm{w}_j^H \bm{h}_{d,k} \right] \label{eq:q_vector_def}
\end{align}
根据定义，矩阵 $\bm{\Psi}$ 为半正定厄米特矩阵（$\bm{\Psi} \succeq \bm{0}$）。

由于约束~\eqref{prob:sub2_unimodular}构成了由 $N$ 个复平面单位圆构成的黎曼流形 $\mathcal{M} = \{\bm{v} \in \mathbb{C}^N : |v_n|=1, \forall n\}$，本节采用复流形共轭梯度法（Riemannian Conjugate Gradient, RCG）进行直接求解：
\begin{enumerate}
  \item \textbf{黎曼梯度投影}：计算欧几里得空间梯度 $\nabla_{\bm{v}} g(\bm{v}) = 2\bm{\Psi}\bm{v} - 2\bm{q}$，并将其切空间投影至复流形 $\mathcal{M}$ 的切空间 $T_{\bm{v}}\mathcal{M}$：
    \begin{equation}
      \mathrm{grad} \, g(\bm{v}) = \nabla_{\bm{v}} g(\bm{v}) - \mathrm{Re}\left\{ \nabla_{\bm{v}} g(\bm{v}) \circ \bm{v}^* \right\} \circ \bm{v}
      \label{eq:riemannian_grad}
    \end{equation}
    其中 $\circ$ 表示 Hadamard 逐元素积。
  \item \textbf{流形回退收缩 (Retraction)}：沿着共轭搜索方向 $\bm{d}$ 进行 Armijo 线搜索步长更新后，通过如下收缩算子将迭代点投影回单位模复圆流形上：
    \begin{equation}
      \mathcal{R}_{\bm{v}}(\alpha \bm{d}) = \left[ \frac{(\bm{v} + \alpha \bm{d})_1}{|(\bm{v} + \alpha \bm{d})_1|}, \dots, \frac{(\bm{v} + \alpha \bm{d})_N}{|(\bm{v} + \alpha \bm{d})_N|} \right]^T
      \label{eq:manifold_retraction}
    \end{equation}
\end{enumerate}
流形共轭梯度法避开了 SDR 升维将变量转化为 $N \times N$ 矩阵的巨大计算开销，直接在 $N$ 维流形上实现超线性收敛。
